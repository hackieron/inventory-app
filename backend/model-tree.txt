User
   │
   └── hasOne Contact
              │
              ├── hasMany Borrowings
              │
              └── hasMany Item Histories


Category
   │
   └── hasMany Items


Location
   │
   └── hasMany Items


Item
   ├── belongsTo Category
   ├── belongsTo Location
   ├── hasMany Borrowings
   ├── hasOne Current Borrowing
   └── hasMany Item Histories


Borrowing
   ├── belongsTo Item
   └── belongsTo Contact


ItemHistory
   ├── belongsTo Item
   └── belongsTo Contact
